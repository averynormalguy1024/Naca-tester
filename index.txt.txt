                        1. NACA Geometry Generation
                    </h3>
                    <p class="mb-2">For a standard NACA 4-Digit Series (e.g., NACA 2412 where m = 0.02, p = 0.4, t = 0.12):</p>
                    <div class="bg-slate-950 p-3 rounded border border-slate-800 font-mono text-cyan-400 overflow-x-auto mb-2">
                        y_t = 5 &middot; t &middot; [0.2969 &middot; &Sqrt;x - 0.1260 &middot; x - 0.3516 &middot; x<sup>2</sup> + 0.2843 &middot; x<sup>3</sup> - 0.1015 &middot; x<sup>4</sup>]
                    </div>
                    <div class="bg-slate-950 p-3 rounded border border-slate-800 font-mono text-cyan-400 overflow-x-auto">
                        y_c = (m / p<sup>2</sup>) &middot; (2px - x<sup>2</sup>) &nbsp; [for x &lt; p] &emsp;|&emsp; y_c = [m / (1-p)<sup>2</sup>] &middot; [(1-2p) + 2px - x<sup>2</sup>] &nbsp; [for x &ge; p]
                    </div>
                </div>

                <div>
                    <h3 class="text-sm font-semibold text-white mb-2 text-cyan-300 border-b border-slate-800 pb-1">
                        2. Potential Circulation & Lift Slope (C<sub>L</sub>)
                    </h3>
                    <div class="bg-slate-950 p-3 rounded border border-slate-800 font-mono text-cyan-400 overflow-x-auto mb-2">
                        C<sub>L, ideal</sub> = 2 &middot; &pi; &middot; (&alpha;<sub>rad</sub> + 2 &middot; m<sub>max</sub>)
                    </div>
                    <p>High angle-of-attack stall is non-linearly modeled as a decay factor f<sub>stall</sub>(&alpha;):</p>
                    <div class="bg-slate-950 p-3 rounded border border-slate-800 font-mono text-cyan-400 overflow-x-auto mt-1">
                        C<sub>L</sub> = C<sub>L, ideal</sub> &middot; max(0.2, 1.0 - ((|&alpha;| - 12) / 8)<sup>2</sup>)
                    </div>
                </div>

                <div>
                    <h3 class="text-sm font-semibold text-white mb-2 text-cyan-300 border-b border-slate-800 pb-1">
                        3. Drag Coefficient Breakdown (C<sub>D</sub>)
                    </h3>
                    <div class="bg-slate-950 p-3 rounded border border-slate-800 font-mono text-cyan-400 overflow-x-auto">
                        C<sub>D,0</sub> = 0.006 + 0.01 &middot; (m<sub>max</sub>)<sup>2</sup> &emsp;|&emsp; C<sub>D,i</sub> = C<sub>L</sub><sup>2</sup> / (&pi; &middot; 8.5) &emsp;|&emsp; C<sub>D</sub> = C<sub>D,0</sub> + C<sub>D,i</sub>
                    </div>
                </div>

                <div>

                        4. Pressure Coefficient (C<sub>p</sub>) & Integrated Forces
                    </h3>
                    <div class="bg-slate-950 p-3 rounded border border-slate-800 font-mono text-cyan-400 overflow-x-auto mb-2">

        function parseNacaCode(code) {
            const N = 80;
            airfoilUpper = [];
            airfoilLower = [];
            airfoilCamber = [];

            // 1. NACA 4-Digit (e.g. 2412, 0015)
            if (/^\d{4}$/.test(code)) {
                const m = parseInt(code[0]) / 100;
                const p = parseInt(code[1]) / 10;
                const t = parseInt(code.substring(2)) / 100;

                for (let i = 0; i <= N; i++) {
                    const x = (1 - Math.cos((i / N) * Math.PI)) / 2;
                    const yt = 5 * t * (0.2969 * Math.sqrt(x) - 0.1260 * x - 0.3516 * x**2 + 0.2843 * x**3 - 0.1015 * x**4);
                    
                    let yc = 0, dyc_dx = 0;
                    if (p > 0) {
                        if (x < p) {
                            yc = (m / (p * p)) * (2 * p * x - x * x);
                            dyc_dx = (2 * m / (p * p)) * (p - x);
                        } else {
                            yc = (m / ((1 - p) ** 2)) * ((1 - 2 * p) + 2 * p * x - x * x);
                            dyc_dx = (2 * m / ((1 - p) ** 2)) * (p - x);
                        }
                    }

                    const theta = Math.atan(dyc_dx);
                    airfoilUpper.push({ x: x - yt * Math.sin(theta), y: yc + yt * Math.cos(theta) });
                    airfoilLower.push({ x: x + yt * Math.sin(theta), y: yc - yt * Math.cos(theta) });
                    airfoilCamber.push({ x: x, y: yc });
                }
                return true;
            }

            // 2. NACA 5-Digit (e.g. 23012, 24015)
            if (/^\d{5}$/.test(code)) {
                const t = parseInt(code.substring(3)) / 100;
                const p = parseInt(code[1]) * 0.05;
                const k1 = 15.893;

                for (let i = 0; i <= N; i++) {
                    const x = (1 - Math.cos((i / N) * Math.PI)) / 2;
                    const yt = 5 * t * (0.2969 * Math.sqrt(x) - 0.1260 * x - 0.3516 * x**2 + 0.2843 * x**3 - 0.1015 * x**4);
                    
                    let yc = 0, dyc_dx = 0;
                    if (x < p) {
                        yc = (k1 / 6) * (x**3 - 3*p*x**2 + p**2*(3-p)*x);
                        dyc_dx = (k1 / 6) * (3*x**2 - 6*p*x + p**2*(3-p));
                    } else {
                        yc = (k1 * p**3 / 6) * (1 - x);
                        dyc_dx = - (k1 * p**3 / 6);
                    }

                    const theta = Math.atan(dyc_dx);
                    airfoilUpper.push({ x: x - yt * Math.sin(theta), y: yc + yt * Math.cos(theta) });
                    airfoilLower.push({ x: x + yt * Math.sin(theta), y: yc - yt * Math.cos(theta) });
                    airfoilCamber.push({ x: x, y: yc });
                }
                return true;
            }

            // 3. NACA 6-Series (e.g. 63-210, 64-412, 65215)
            const sixPattern = /^6\d[\- ]?\d{3}$/;
            if (sixPattern.test(code)) {
                const clean = code.replace(/[\- ]/g, '');
                const t = parseInt(clean.substring(3)) / 100;

                for (let i = 0; i <= N; i++) {
                    const x = (1 - Math.cos((i / N) * Math.PI)) / 2;
                    const yt = 5 * t * (0.2969 * Math.sqrt(x) - 0.08 * x - 0.35 * x**2 + 0.18 * x**3 - 0.05 * x**4);
                    const yc = 0.018 * Math.sin(Math.PI * Math.pow(x, 0.85));
                    const dyc_dx = 0.018 * Math.PI * 0.85 * Math.pow(x, -0.15) * Math.cos(Math.PI * Math.pow(x, 0.85));

                    const theta = Math.atan(dyc_dx || 0);
                    airfoilUpper.push({ x: x - yt * Math.sin(theta), y: yc + yt * Math.cos(theta) });
                    airfoilLower.push({ x: x + yt * Math.sin(theta), y: yc - yt * Math.cos(theta) });
                    airfoilCamber.push({ x: x, y: yc });
                }
                return true;
            }

            return false;
        }

        function updateCalculationsAndCharts() {
            const radAlpha = (alpha * Math.PI) / 180;
            
            let maxCamber = 0;
            airfoilCamber.forEach(pt => { if (Math.abs(pt.y) > maxCamber) maxCamber = Math.abs(pt.y); });

            // Tunnel Constriction Boost Factor
            const tunnelFactor = isWindTunnel ? 1.15 : 1.0;
            const effectiveV = vInf * tunnelFactor;

            const Cl_ideal = 2 * Math.PI * (radAlpha + 2 * maxCamber);
            
            let stallFactor = 1.0;
            if (alpha > 12) stallFactor = Math.max(0.2, 1.0 - Math.pow((alpha - 12) / 8, 2));
            else if (alpha < -12) stallFactor = Math.max(0.2, 1.0 - Math.pow((-alpha - 12) / 8, 2));

            const Cl = Cl_ideal * stallFactor * (isWindTunnel ? 1.08 : 1.0);
            const Cd0 = 0.006 + 0.01 * Math.pow(maxCamber, 2);
            const Cd = (Cd0 + Math.pow(Cl, 2) / (Math.PI * 8.5)) * (isWindTunnel ? 1.12 : 1.0);
            const LdRatio = Cd > 0 ? (Cl / Cd) : 0;

            const rho = 1.225;
            const q = 0.5 * rho * effectiveV * effectiveV;
            const liftForce = Cl * q;
            const dragForce = Cd * q;

            const Xcp = 0.25 - (0.1 * (Cl - 0.2));
            const Cm = -0.05 - 0.2 * maxCamber;

            document.getElementById('hudCl').innerText = Cl.toFixed(3);
            document.getElementById('hudCd').innerText = Cd.toFixed(4);
            document.getElementById('hudLd').innerText = LdRatio.toFixed(1);

            document.getElementById('calcLift').innerText = liftForce.toFixed(1) + ' N/m';
            document.getElementById('calcDrag').innerText = dragForce.toFixed(2) + ' N/m';
            document.getElementById('calcXcp').innerText = Xcp.toFixed(3) + ' c';
            document.getElementById('calcCm').innerText = Cm.toFixed(3);

            updateChartsData(Cl, maxCamber);
        }

        // Potential Flow & Tunnel Boundary Evaluator
        function getFlowVelocityAt(x, y) {
            const radAlpha = (alpha * Math.PI) / 180;
            const tunnelSpeed = isWindTunnel ? vInf * 1.12 : vInf;
            
            let u = tunnelSpeed * Math.cos(radAlpha);
            let v = tunnelSpeed * Math.sin(radAlpha);

            const rx = x - 0.25;
            const ry = y;
            const r2 = rx * rx + ry * ry + 0.001;

            let maxCamber = 0;
            airfoilCamber.forEach(pt => { if (Math.abs(pt.y) > maxCamber) maxCamber = Math.abs(pt.y); });
            const gamma = 0.5 * tunnelSpeed * Math.PI * (radAlpha + 2 * maxCamber);

            u += (gamma * ry) / (2 * Math.PI * r2);
            v -= (gamma * rx) / (2 * Math.PI * r2);

            // Wind Tunnel Top/Bottom Wall Bounding Constraint
            if (isWindTunnel) {
                const wallDist = 0.45;
                if (Math.abs(y) > wallDist - 0.05) {
                    v *= 0.1; // Bounded stream lines
                }
            }

            const speed = Math.sqrt(u * u + v * v);
            const cp = 1 - (speed * speed) / (tunnelSpeed * tunnelSpeed);

            return { u, v, speed, cp };
        }

        function initParticles() {
            particles.length = 0;
            for (let i = 0; i < NUM_PARTICLES; i++) {
                particles.push({

                    y: (Math.random() * 1.0) - 0.5,
                    life: Math.random() * 100,
                    history: []
                });
            }
        }

        let canvas, ctx;
        function initCanvas() {
            canvas = document.getElementById('flowCanvas');
            ctx = canvas.getContext('2d');

            function resize() {
                canvas.width = canvas.parentElement.clientWidth;
                canvas.height = canvas.parentElement.clientHeight;
            }
            window.addEventListener('resize', resize);
            resize();

            canvas.addEventListener('mousedown', (e) => {
                isDragging = true;
                lastMouseX = e.clientX;
                lastMouseY = e.clientY;
            });

            window.addEventListener('mouseup', () => isDragging = false);
            canvas.addEventListener('mousemove', (e) => {
                if (isDragging) {
                    panX += e.clientX - lastMouseX;
                    panY += e.clientY - lastMouseY;
                    lastMouseX = e.clientX;
                    lastMouseY = e.clientY;
                }
            });

            canvas.addEventListener('wheel', (e) => {
                e.preventDefault();
                zoom *= e.deltaY < 0 ? 1.1 : 0.9;
                zoom = Math.max(0.4, Math.min(zoom, 4.0));
            });
        }

        function resetZoom() {
            panX = 0; panY = 0; zoom = 1.0;
        }

        function worldToScreen(wx, wy) {
            const scale = Math.min(canvas.width, canvas.height) * 0.55 * zoom;
            const cx = canvas.width / 2 + panX;
            const cy = canvas.height / 2 + panY;
            return { x: cx + (wx - 0.35) * scale, y: cy - wy * scale };
        }

        function renderLoop() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            const showHeatmap = document.getElementById('chkHeatmap').checked;
            const showStreamlines = document.getElementById('chkStreamlines').checked;
            const showParticles = document.getElementById('chkParticles').checked;
            const showVectors = document.getElementById('chkVectors').checked;

            // 1. Heatmap Field
            if (showHeatmap) {
                const gridX = 42, gridY = 24;
                const cellW = canvas.width / gridX, cellH = canvas.height / gridY;

                for (let i = 0; i < gridX; i++) {
                    for (let j = 0; j < gridY; j++) {
                        const px = i * cellW + cellW / 2, py = j * cellH + cellH / 2;
                        const scale = Math.min(canvas.width, canvas.height) * 0.55 * zoom;
                        const cx = canvas.width / 2 + panX, cy = canvas.height / 2 + panY;
                        
                        const wx = (px - cx) / scale + 0.35;
                        const wy = -(py - cy) / scale;

                        const flow = getFlowVelocityAt(wx, wy);
                        const norm = Math.max(0, Math.min(1, (flow.cp + 1.2) / 2.2));

                        const r = Math.round(Math.min(255, Math.max(0, (norm - 0.5) * 2 * 255)));
                        const g = Math.round(Math.min(255, Math.max(0, (1 - Math.abs(norm - 0.5) * 2) * 200)));
                        const b = Math.round(Math.min(255, Math.max(0, (0.5 - norm) * 2 * 255)));

                        ctx.fillStyle = `rgba(${r}, ${g}, ${b}, 0.16)`;
                        ctx.fillRect(i * cellW, j * cellH, cellW + 1, cellH + 1);
                    }
                }
            }

            // 2. Wind Tunnel Wall Render
            if (isWindTunnel) {
                const topWall = worldToScreen(0, 0.45);
                const botWall = worldToScreen(0, -0.45);

                // Upper Wall
                ctx.fillStyle = 'rgba(245, 158, 11, 0.15)';
                ctx.fillRect(0, 0, canvas.width, topWall.y);
                ctx.strokeStyle = '#f59e0b';
                ctx.lineWidth = 3;
                ctx.beginPath();
                ctx.moveTo(0, topWall.y); ctx.lineTo(canvas.width, topWall.y);
                ctx.stroke();

                // Lower Wall
                ctx.fillRect(0, botWall.y, canvas.width, canvas.height - botWall.y);
                ctx.beginPath();
                ctx.moveTo(0, botWall.y); ctx.lineTo(canvas.width, botWall.y);
                ctx.stroke();

                // Honeycomb Flow Injector Lines
                ctx.strokeStyle = 'rgba(245, 158, 11, 0.3)';
                ctx.lineWidth = 1;
                for (let y = topWall.y; y <= botWall.y; y += 15) {
                    ctx.beginPath();
                    ctx.moveTo(0, y); ctx.lineTo(25, y);
                    ctx.stroke();
                }
            }

            // 3. Vector Grid
            if (showVectors) {
                ctx.strokeStyle = 'rgba(148, 163, 184, 0.25)';
                ctx.lineWidth = 1;

                for (let x = -0.6; x <= 1.3; x += 0.12) {
                    for (let y = -0.4; y <= 0.4; y += 0.12) {
                        const flow = getFlowVelocityAt(x, y);
                        const p1 = worldToScreen(x, y);
                        const arrowLen = 12 * (flow.speed / vInf);
                        const angle = Math.atan2(-flow.v, flow.u);

                        ctx.beginPath();
                        ctx.moveTo(p1.x, p1.y);
                        ctx.lineTo(p1.x + arrowLen * Math.cos(angle), p1.y + arrowLen * Math.sin(angle));
                        ctx.stroke();
                    }
                }
            }

            // 4. Streamlines
            if (showStreamlines) {
                ctx.lineWidth = 1;
                const yBound = isWindTunnel ? 0.4 : 0.55;
                for (let yStart = -yBound; yStart <= yBound; yStart += 0.08) {
                    ctx.strokeStyle = isWindTunnel ? 'rgba(245, 158, 11, 0.25)' : 'rgba(56, 189, 248, 0.2)';
                    ctx.beginPath();

                    let currX = -0.8, currY = yStart;
                    let pt = worldToScreen(currX, currY);
                    ctx.moveTo(pt.x, pt.y);

                    for (let step = 0; step < 75; step++) {
                        const flow = getFlowVelocityAt(currX, currY);
                        const dt = 0.02;
                        currX += (flow.u / vInf) * dt;
                        currY += (flow.v / vInf) * dt;

                        pt = worldToScreen(currX, currY);
                        ctx.lineTo(pt.x, pt.y);
                    }
                    ctx.stroke();
                }
            }

            // 5. Particles
            if (showParticles) {
                const dt = 0.015;
                particles.forEach(p => {
                    const flow = getFlowVelocityAt(p.x, p.y);
                    
                    p.x += (flow.u / vInf) * dt * (vInf / 40);
                    p.y += (flow.v / vInf) * dt * (vInf / 40);
                    p.life += 1;

                    p.history.push({ ...worldToScreen(p.x, p.y) });
                    if (p.history.length > 7) p.history.shift();

                    if (p.x > 1.5 || p.y < -0.6 || p.y > 0.6 || p.life > 160) {
                        p.x = -0.8;
                        p.y = (Math.random() * (isWindTunnel ? 0.8 : 1.2)) - (isWindTunnel ? 0.4 : 0.6);
                        p.life = 0;
                        p.history = [];
                    }

                    if (p.history.length > 1) {
                        ctx.beginPath();
                        ctx.moveTo(p.history[0].x, p.history[0].y);
                        for (let k = 1; k < p.history.length; k++) {
                            ctx.lineTo(p.history[k].x, p.history[k].y);
                        }
                        
                        ctx.strokeStyle = isWindTunnel ? 'rgba(251, 191, 36, 0.5)' : 'rgba(56, 189, 248, 0.5)';
                        ctx.lineWidth = 1.8;
                        ctx.stroke();
                    }
                });
            }

            // 6. Airfoil Shape Render
            if (airfoilUpper.length > 0) {
                const radAlpha = (alpha * Math.PI) / 180;
                
                const rotatePt = (pt) => {
                    const rx = pt.x - 0.25;
                    const ry = pt.y;
                    const xRot = rx * Math.cos(-radAlpha) - ry * Math.sin(-radAlpha) + 0.25;
                    const yRot = rx * Math.sin(-radAlpha) + ry * Math.cos(-radAlpha);
                    return worldToScreen(xRot, yRot);
                };

                ctx.beginPath();
                const p0 = rotatePt(airfoilUpper[0]);
                ctx.moveTo(p0.x, p0.y);

                for (let i = 1; i < airfoilUpper.length; i++) {
                    const p = rotatePt(airfoilUpper[i]);
                    ctx.lineTo(p.x, p.y);
                }
                for (let i = airfoilLower.length - 1; i >= 0; i--) {
                    const p = rotatePt(airfoilLower[i]);
                    ctx.lineTo(p.x, p.y);
                }
                ctx.closePath();

                ctx.fillStyle = '#030712';
                ctx.fill();
                ctx.strokeStyle = '#38bdf8';
                ctx.lineWidth = 2.5;
                ctx.stroke();
            }

            requestAnimationFrame(renderLoop);
        }

        function updateChartsData(currentCl, maxCamber) {
            if (!cpChartInstance) return;

            const xVals = Array.from({length: 20}, (_, i) => i / 19);
            const upperCp = [], lowerCp = [], velRatio = [];

            xVals.forEach(x => {
                const radAlpha = (alpha * Math.PI) / 180;
                const cpUp = -1.2 * Math.sin(Math.PI * x) * (1 + radAlpha) - (currentCl * 0.5);
                const cpLow = 0.6 * Math.sin(Math.PI * x) - (currentCl * 0.2);
                
                upperCp.push(cpUp);
                lowerCp.push(cpLow);
                velRatio.push(Math.sqrt(Math.max(0, 1 - cpUp)));
            });

            cpChartInstance.data.datasets[0].data = upperCp;
            cpChartInstance.data.datasets[1].data = lowerCp;
            cpChartInstance.update();

            velChartInstance.data.datasets[0].data = velRatio;
            velChartInstance.update();

            const alphaRange = Array.from({length: 21}, (_, i) => -10 + i);
            const clCurve = alphaRange.map(a => {
                const r = (a * Math.PI) / 180;
                let factor = 1.0;
                if (a > 12) factor = Math.max(0.2, 1.0 - Math.pow((a - 12) / 8, 2));
                if (a < -12) factor = Math.max(0.2, 1.0 - Math.pow((-a - 12) / 8, 2));
                return 2 * Math.PI * (r + 2 * maxCamber) * factor;
            });

            clAlphaChartInstance.data.datasets[0].data = clCurve;
            const currentIdx = alphaRange.indexOf(Math.round(alpha));
            const markerData = new Array(21).fill(null);
            if (currentIdx !== -1) markerData[currentIdx] = currentCl;
            clAlphaChartInstance.data.datasets[1].data = markerData;
            clAlphaChartInstance.update();

            const cdCurve = clCurve.map(cl => 0.006 + Math.pow(cl, 2) / (Math.PI * 8.5));
            polarChartInstance.data.labels = cdCurve.map(cd => cd.toFixed(3));
            polarChartInstance.data.datasets[0].data = clCurve;
            polarChartInstance.update();
        }
